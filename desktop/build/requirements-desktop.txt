# Desktop backend dependencies (slimmed down from the full requirements).
#
# The desktop client talks to the web channel and the agent core. Model calls
# go through the LiteLLM HTTP API, so vendor SDKs are not installed. Messaging
# channel packages are not installed.

# cryptography stopped shipping macOS x86_64 wheels at 49.0. A source build
# links the runner's Homebrew OpenSSL, and PyInstaller then bundles it beside
# Python's older libssl.3.dylib under the same name, so the extension fails to
# load on users' machines. Wheels link OpenSSL statically; forcing them makes
# pip settle on the newest version that has one per arch.
--only-binary=cryptography

# ---- core ----
numpy>=1.21
# aiohttp<3.10 has no prebuilt wheels for Python 3.13+ (would need MSVC to build).
aiohttp>=3.8.6,<3.10; python_version < "3.13"
aiohttp>=3.10; python_version >= "3.13"
requests>=2.28.2
chardet>=5.1.0
Pillow
python-dotenv>=1.0.0
PyYAML>=6.0
croniter>=2.0.0
# zoneinfo is stdlib from 3.9 (the Win7 build runs 3.8); Windows has no system tz database
backports.zoneinfo; python_version < "3.9"
tzdata; sys_platform == "win32"
click>=8.0
json-repair

# ---- web framework (web channel) ----
# web.py 0.62 (the last release supporting Python < 3.10) imports the stdlib cgi
# module removed in 3.13; 0.76 dropped it but requires 3.10+.
web.py; python_version < "3.10"
web.py>=0.76; python_version >= "3.10"

# ---- token counting ----
tiktoken>=0.3.2

# ---- voice audio container (the speech HTTP calls use LiteLLM) ----
pydub>=0.25.1

# ---- document parsing (web_fetch / knowledge) ----
pypdf
python-docx
openpyxl
python-pptx

# ---- browser tool ----
# Only the pure-Python package + Node driver are bundled by PyInstaller (~10-15MB).
# The Chromium binary is NOT bundled: the browser tool drives the user's system
# Chrome/Edge, or downloads Chromium on demand into ~/.maia at first use.
playwright==1.52.0
